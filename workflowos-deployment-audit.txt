async function queryOne(sql, params = []) {
  const [rows] = await db.execute(sql, params);
  return rows[0] || null;
}

function isDuplicateError(error) {
  return error?.code === 'ER_DUP_ENTRY' || error?.errno === 1062;
}

function asyncHandler(handler) {
  return (req, res, next) => {
    Promise.resolve(handler(req, res, next)).catch(next);
  };
}

function getToken(req) {
  const header = req.headers.authorization || '';
  return header.startsWith('Bearer ') ? header.slice(7) : null;
}

async function getSessionFromRequest(req) {
  const token = getToken(req);

  if (!token) return null;

  return queryOne(
    `SELECT *
     FROM sessions
     WHERE id = ?
       AND expires_at > UTC_TIMESTAMP()`,
    [token]
  );
}

async function requireAuth(req, res, next) {
  try {
    const session = await getSessionFromRequest(req);

    if (!session) {
      return res.status(401).json({ error: 'Unauthorized' });
    }

    req.user = { id: session.user_id };
    req.businessId = session.business_id;

    return next();
  } catch (error) {
    return next(error);
  }
}

async function requireBusinessAccess(req, res, next) {
  try {
    const businessId = Number(
      req.params.businessId || req.body?.businessId || req.businessId
    );

    if (!businessId) {
      return res.status(400).json({ error: 'Missing business context' });
    }

    const match = await getCurrentBusinessForUser(
      req.user.id,
      businessId
    );

    if (!match) {
      return res.status(403).json({ error: 'Forbidden' });
    }

    req.businessId = businessId;

    return next();
  } catch (error) {
    return next(error);
  }
}

async function isCatalogueAdmin(userId) {
  const allowedEmails = String(
    process.env.CATALOGUE_ADMIN_EMAILS || ''
  )
    .split(',')
    .map((email) => email.trim().toLowerCase())
    .filter(Boolean);

  const user = await queryOne(
    'SELECT email FROM users WHERE id = ?',
    [userId]
  );

  return Boolean(
    user && allowedEmails.includes(user.email.toLowerCase())
  );
}

async function requireCatalogueAdmin(req, res, next) {
  try {
    if (!(await isCatalogueAdmin(req.user.id))) {
      return res.status(403).json({
        error: 'Platform catalogue administrator access required',
      });
    }

    return next();
  } catch (error) {
    return next(error);
  }
}

app.post('/api/register', asyncHandler(async (req, res) => {
  const {
    firstName,
    lastName,
    email,
    password,
    businessName,
    businessType,
  } = req.body || {};

  if (
    !firstName ||
    !lastName ||
    !email ||
    !password ||
    !businessName ||
    !businessType
  ) {
    return res.status(400).json({
      error: 'Missing required fields',
    });
  }

  try {
    const user = await createUser({
      firstName,
      lastName,
      email,
      password,
    });

    const business = await createBusiness({
      userId: user.id,
      name: businessName,
      businessType,
      currency: 'ZAR',
      timezone: 'Africa/Johannesburg',
    });

    const { sessionId } = await createSession(
      user.id,
      business.id
    );

    return res.status(201).json({
      user: {
        id: user.id,
        firstName: user.first_name,
        lastName: user.last_name,
        email: user.email,
      },
      business: {
        id: business.id,
        name: business.name,
        businessType: business.business_type,
      },
      sessionToken: sessionId,
    });
  } catch (error) {
    if (error.message === 'USER_EXISTS' || isDuplicateError(error)) {
      return res.status(409).json({
        error: 'User already exists',
      });
    }

    console.error('Registration error:', error);

    return res.status(500).json({
      error: 'Registration failed',
    });
  }
}));

app.post('/api/login', asyncHandler(async (req, res) => {
  const { email, password } = req.body || {};

  if (!email || !password) {
    return res.status(400).json({
      error: 'Email and password required',
    });
  }

  const user = await getUserByEmail(email);

  if (!user || !verifyPassword(password, user.password_hash)) {
    return res.status(401).json({
      error: 'Invalid credentials',
    });
  }

  const memberships = await getBusinessMembershipsForUser(user.id);
  const business = memberships[0] || null;

  const { sessionId } = await createSession(
    user.id,
    business ? business.business_id : null
  );

  return res.json({
    user: {
      id: user.id,
      firstName: user.first_name,
      lastName: user.last_name,
      email: user.email,
    },
    business: business
      ? {
          id: business.business_id,
          name: business.business_name,
        }
      : null,
    sessionToken: sessionId,
  });
}));

app.get('/api/session', requireAuth, asyncHandler(async (req, res) => {
  const memberships = await getBusinessMembershipsForUser(req.user.id);

  const activeBusiness =
    memberships.find(
      (item) => item.business_id === Number(req.businessId)
    ) ||
    memberships[0] ||
    null;

  const user = await getUserById(req.user.id);

  if (!user) {
    return res.status(404).json({ error: 'User not found' });
  }

  return res.json({
    user: {
      id: req.user.id,
      email: user.email,
      firstName: user.first_name,
      lastName: user.last_name,
    },
    business: activeBusiness
      ? {
          id: activeBusiness.business_id,
          name: activeBusiness.business_name,
          businessType: activeBusiness.business_type,
          currency: activeBusiness.currency,
          timezone: activeBusiness.timezone,
        }
      : null,
    memberships: memberships.map((row) => ({
      id: row.business_id,
      name: row.business_name,
      role: row.role,
    })),
  });
}));

function startServer(port = PORT) {
  return new Promise((resolve, reject) => {
    const server = app.listen(port, () => {
      const actualPort = server.address().port;

      console.log(
        `WorkflowOS listening on http://localhost:${actualPort}`
      );

      resolve(server);
    });

    server.on('error', (error) => {
      if (error.code === 'EADDRINUSE') {
        const nextPort = Number(port) + 1;

        console.warn(
          `Port ${port} is busy. Retrying on ${nextPort}`
        );

        server.close();
        resolve(startServer(nextPort));
        return;
      }

      reject(error);
    });
  });
}

async function withTransaction(callback) {
  const connection = await db.getConnection();

  try {
    await connection.beginTransaction();
    const result = await callback(connection);
    await connection.commit();
    return result;
  } catch (error) {
    try {
      await connection.rollback();
    } catch (rollbackError) {
      console.error('Transaction rollback failed:', rollbackError.message);
    }

    throw error;
  } finally {
    connection.release();
  }
}

function normalizeImageUrl(value) {
  const imageUrl = String(value || '').trim();

  if (!imageUrl) {
    return null;
  }

  let parsedImageUrl;

  try {
    parsedImageUrl = new URL(imageUrl);
  } catch {
    throw new Error('INVALID_IMAGE_URL');
  }

  if (
    parsedImageUrl.protocol !== 'https:' ||
    !parsedImageUrl.hostname ||
    imageUrl.length > 1000
  ) {
    throw new Error('INVALID_IMAGE_URL');
  }

  return imageUrl;
}

const PRODUCT_SELECT = `
  SELECT
    p.id,
    p.business_id,
    p.name,
    p.brand,
    COALESCE(
      p.barcode,
      (
        SELECT pb.barcode
        FROM product_barcodes pb
        WHERE pb.global_product_id = p.global_catalogue_id
          AND pb.verification_status = 'verified'
        ORDER BY pb.id
        LIMIT 1
      )
    ) AS barcode,
    CASE
      WHEN p.barcode_status = 'verified'
        OR EXISTS (
          SELECT 1
          FROM product_barcodes pb
          WHERE pb.global_product_id = p.global_catalogue_id
            AND pb.verification_status = 'verified'
        )
      THEN 'verified'
      ELSE 'unverified'
    END AS barcode_status,
    p.description,
    p.packaging_size,
    p.unit_of_measure,
    p.image_url,
    p.sku,
    p.category,
    p.price_cents,
    p.cost_cents,
    p.cost_is_known,
    p.is_active,
    p.created_at,
    p.updated_at
`;

async function createProduct({
  businessId,
  userId,
  payload = {},
}) {
  const membership = await getOne(
    `SELECT id FROM business_memberships
     WHERE business_id = ? AND user_id = ?`,
    [businessId, userId]
  );

  if (!membership) {
    throw new Error('FORBIDDEN');
  }

  const sanitizedName = String(payload.name || '').trim();
  const sanitizedSku = String(payload.sku || '').trim();

  const priceCents = Number(payload.price || 0) * 100;

  const costKnown =
    payload.cost !== undefined &&
    payload.cost !== null &&
    payload.cost !== '';

  const costCents = Number(payload.cost || 0) * 100;

  const initialStock = Number(payload.initialStock || 0);
  const reorderLevel = Number(payload.reorderLevel || 0);

  const unitOfMeasure = String(
    payload.unitOfMeasure || 'unit'
  ).trim();

  const imageUrl = normalizeImageUrl(payload.imageUrl || '');

  let barcode = null;

  if (payload.barcode) {
    try {
      barcode = normalizeBarcode(payload.barcode).barcode;
    } catch {
      throw new Error('INVALID_BARCODE');
    }
  }

  if (
    !sanitizedName ||
    sanitizedName.length > 180 ||
    !Number.isFinite(priceCents) ||
    priceCents < 0 ||
    !Number.isSafeInteger(Math.round(priceCents)) ||
    !Number.isFinite(costCents) ||
    costCents < 0 ||
    !Number.isSafeInteger(Math.round(costCents)) ||
    !Number.isSafeInteger(initialStock) ||
    initialStock < 0 ||
    !Number.isSafeInteger(reorderLevel) ||
    reorderLevel < 0 ||
    !unitOfMeasure ||
    unitOfMeasure.length > 32 ||
    (imageUrl && imageUrl.length > 1000)
  ) {
    throw new Error('INVALID_PRODUCT');
  }

  const productId = await withTransaction(async (connection) => {
    const result = await execute(
      `INSERT INTO products
       (
         business_id,
         name,
         brand,
         barcode,
         barcode_status,
         description,
         packaging_size,
         unit_of_measure,
         image_url,
         sku,
         category,
         price_cents,
         cost_cents,
         is_active,
         cost_is_known
       )
       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        businessId,
        sanitizedName,
        String(payload.brand || '').trim() || null,
        barcode,
        'unverified',
        String(payload.description || '').trim() || null,
        String(payload.packagingSize || '').trim() || null,
        unitOfMeasure,
        imageUrl,
        sanitizedSku || null,
        payload.category || null,
        Math.round(priceCents),
        Math.round(costCents),
        payload.isActive === false ? 0 : 1,
        costKnown ? 1 : 0,
      ],
      connection
    );

    const id = result.insertId;
    const now = new Date();

    await execute(
      `INSERT INTO inventory
       (business_id, product_id, quantity_on_hand, reorder_level, updated_at)
       VALUES (?, ?, ?, ?, ?)`,
      [businessId, id, initialStock, reorderLevel, now],
      connection
    );

    if (initialStock > 0) {
      await execute(
        `INSERT INTO inventory_movements
         (
           business_id,
           product_id,
           movement_type,
           quantity,
           reference_type,
           reference_id,
           notes,
           created_at
         )
         VALUES (?, ?, 'opening', ?, 'product', ?, 'Opening stock', ?)`,
        [businessId, id, initialStock, id, now],
        connection
      );
    }

    return id;
  });

  return getProductById(productId);
}

async function updateProduct({

  const membership = await getOne(

     WHERE business_id = ? AND user_id = ?`,

  if (

    Number(productId) <= 0
  ) {
    throw new Error('PRODUCT_NOT_FOUND');
  }

  const current = await getOne(
    `SELECT * FROM products
     WHERE id = ? AND business_id = ?`,
    [productId, businessId]
  );

  if (!current) {
    throw new Error('PRODUCT_NOT_FOUND');
  }

  const inventory = await getOne(
    `SELECT quantity_on_hand, reorder_level
     FROM inventory
     WHERE business_id = ? AND product_id = ?`,
    [businessId, productId]
  );

  if (!inventory) {
    throw new Error('PRODUCT_NOT_FOUND');
  }

  const name =
    payload.name === undefined
      ? current.name
      : payload.name === null
        ? ''
        : String(payload.name).trim();

  const optionalText = (value) =>
    value === null || value === undefined

async function createSale({
  businessId,
  userId,
  payload = {},
  idempotencyKey,
}) {
  const membership = await getOne(
    `SELECT id FROM business_memberships
     WHERE business_id = ? AND user_id = ?`,
    [businessId, userId]
  );

  if (!membership) {
    throw new Error('FORBIDDEN');
  }

  if (!Array.isArray(payload.items) || payload.items.length === 0) {
    throw new Error('INVALID_SALE');
  }

  if (payload.items.length > 100) {
    throw new Error('INVALID_SALE');
  }

  try {
    const transactionResult = await withTransaction(async (connection) => {
      if (idempotencyKey) {
        const existing = await getOne(
          `SELECT * FROM sales
           WHERE business_id = ? AND idempotency_key = ?
           FOR UPDATE`,
          [businessId, idempotencyKey],
          connection
        );

        if (existing) {
          return {
            saleId: existing.id,
            replayed: true,
          };
        }
      }

      const customerId = payload.customerId
        ? Number(payload.customerId)
        : null;

      if (
        customerId &&
        !(await getOne(
          `SELECT id FROM customers
           WHERE id = ? AND business_id = ?`,
          [customerId, businessId],
          connection
        ))
      ) {
        throw new Error('INVALID_CUSTOMER');
      }

      const items = [];

      for (const item of payload.items) {
        const productId = Number(item.productId);
        const quantity = Number(item.quantity);

        if (
          !Number.isSafeInteger(productId) ||
          !Number.isSafeInteger(quantity) ||
          quantity <= 0
        ) {
          throw new Error('INVALID_SALE');
        }

        const product = await getOne(
          `SELECT
             p.id,
             p.business_id,
             p.name,
             p.category,
             p.price_cents,
             p.cost_cents,
             p.cost_is_known,
             i.quantity_on_hand
           FROM products p
           JOIN inventory i
             ON i.product_id = p.id
             AND i.business_id = p.business_id
           WHERE p.id = ?
             AND p.business_id = ?
             AND p.is_active = 1
           FOR UPDATE`,
          [productId, businessId],
          connection
        );

        if (!product) {
          throw new Error('INVALID_PRODUCT');
        }

        if (Number(product.quantity_on_hand) < quantity) {
          throw new Error('INSUFFICIENT_STOCK');
        }

        items.push({
          product,
          quantity,
          lineTotalCents:
            Number(product.price_cents) * quantity,
          unitCostCents: product.cost_is_known
            ? Number(product.cost_cents)
            : null,
        });
      }

      const subtotalCents = items.reduce(
        (sum, item) => sum + item.lineTotalCents,
        0
      );

      if (
        !items.every((item) =>
          Number.isSafeInteger(item.lineTotalCents)
        ) ||
        !Number.isSafeInteger(subtotalCents)
      ) {
        throw new Error('INVALID_SALE');
      }

      const now = new Date();
      const reference = `WOS-${crypto.randomUUID()}`;

      const saleResult = await execute(
        `INSERT INTO sales
         (
           business_id,
           customer_id,
           transaction_reference,
           idempotency_key,
           subtotal_cents,
           discount_cents,
           tax_cents,
           total_cents,
           payment_method,
           status,
           created_at,
           updated_at
         )
         VALUES (?, ?, ?, ?, ?, 0, 0, ?, ?, 'completed', ?, ?)`,
        [
          businessId,
          customerId,
          reference,
          idempotencyKey || null,
          subtotalCents,
          subtotalCents,
          payload.paymentMethod || null,
          now,
          now,
        ],
        connection
      );

      const saleId = saleResult.insertId;

      for (const item of items) {
        await execute(
          `INSERT INTO sale_items
           (
             sale_id,
             product_id,
             product_name_snapshot,
             product_category_snapshot,
             quantity,
             unit_price_cents,
             unit_cost_cents,
             line_total_cents
           )
           VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
          [
            saleId,
            item.product.id,
            item.product.name,
            item.product.category,
            item.quantity,
            item.product.price_cents,
            item.unitCostCents,
            item.lineTotalCents,
          ],
          connection
        );

        const stockUpdate = await execute(
          `UPDATE inventory
           SET quantity_on_hand = quantity_on_hand - ?,
               updated_at = ?
           WHERE business_id = ?
             AND product_id = ?
             AND quantity_on_hand >= ?`,
          [
            item.quantity,
            now,
            businessId,
            item.product.id,
            item.quantity,
          ],
          connection
        );

        if (stockUpdate.affectedRows !== 1) {
          throw new Error('INSUFFICIENT_STOCK');
        }

        await execute(
          `INSERT INTO inventory_movements
           (
             business_id,
             product_id,
             movement_type,
             quantity,
             reference_type,
             reference_id,
             notes,
             created_at
           )
           VALUES (?, ?, 'sale', ?, 'sale', ?, 'Stock sold', ?)`,
          [
            businessId,
            item.product.id,
            -item.quantity,
            saleId,
            now,
          ],
          connection
        );
      }

      return {
        saleId,
        replayed: false,
      };
    });

    const sale = await getOne(
      'SELECT * FROM sales WHERE id = ?',
      [transactionResult.saleId]
    );

    return {
      sale,
      replayed: transactionResult.replayed,
    };
  } catch (error) {
    if (isDuplicateEntry(error) && idempotencyKey) {
      const existing = await getOne(
        `SELECT * FROM sales
         WHERE business_id = ? AND idempotency_key = ?`,
        [businessId, idempotencyKey]
      );

      if (existing) {
        return {
          sale: existing,
          replayed: true,
        };
      }
    }

    throw error;
  }
}

async function refundSale({ businessId, saleId }) {
  await withTransaction(async (connection) => {
    const sale = await getOne(
      `SELECT * FROM sales
       WHERE id = ? AND business_id = ?
       FOR UPDATE`,
      [saleId, businessId],
      connection
    );

    if (!sale) {
      throw new Error('SALE_NOT_FOUND');
    }

    if (sale.status !== 'completed') {
      throw new Error('SALE_NOT_REFUNDABLE');
    }

    const now = new Date();

    const update = await execute(
      `UPDATE sales
       SET status = 'refunded',
           updated_at = ?
       WHERE id = ?
         AND business_id = ?
         AND status = 'completed'`,
      [now, saleId, businessId],
      connection
    );

    if (update.affectedRows !== 1) {
      throw new Error('SALE_NOT_REFUNDABLE');
    }

    const items = await getMany(
      `SELECT product_id, quantity
       FROM sale_items
       WHERE sale_id = ?`,
      [saleId],
      connection
    );

    for (const item of items) {
      await execute(
        `UPDATE inventory
         SET quantity_on_hand = quantity_on_hand + ?,
             updated_at = ?
         WHERE business_id = ?
           AND product_id = ?`,
        [
          item.quantity,
          now,
          businessId,
          item.product_id,
        ],
        connection
      );

      await execute(
        `INSERT INTO inventory_movements
         (
           business_id,
           product_id,
           movement_type,
           quantity,
           reference_type,
           reference_id,
           notes,
           created_at
         )
         VALUES (?, ?, 'refund', ?, 'sale', ?, 'Refund stock reversal', ?)`,
        [
          businessId,
          item.product_id,
          item.quantity,
          saleId,
          now,
        ],
        connection
      );
    }
  });

  return getOne(
    'SELECT * FROM sales WHERE id = ?',
    [saleId]
  );
}

async function adjustInventory({
  businessId,
  userId,
  productId,
  quantity,
  reorderLevel,
  notes,
}) {
  const membership = await getOne(
    `SELECT id FROM business_memberships
     WHERE business_id = ? AND user_id = ?`,
    [businessId, userId]
  );

  if (!membership) {
    throw new Error('FORBIDDEN');
  }

  if (
    !Number.isSafeInteger(quantity) ||

    (

      (
        !Number.isSafeInteger(reorderLevel) ||
        reorderLevel < 0
      )
    )
  ) {
    throw new Error('INVALID_ADJUSTMENT');

  await withTransaction(async (connection) => {

    if (
      Number(inventoryRow.quantity_on_hand) + quantity < 0

      throw new Error('INSUFFICIENT_STOCK');

      `UPDATE inventory

           updated_at = ?

      `INSERT INTO inventory_movements
       (
         business_id,
         product_id,
         movement_type,
         quantity,
         reference_type,
         reference_id,
         notes,
         created_at
       )
       VALUES (?, ?, ?, ?, 'adjustment', NULL, ?, ?)`,
      [
        businessId,
        productId,
        quantity > 0 ? 'receipt' : 'adjustment',
        quantity,
        String(notes || '').slice(0, 250) || null,
        now,
      ],
      connection
    );
  });

  return getOne(
    `SELECT * FROM inventory
     WHERE business_id = ?
       AND product_id = ?`,
    [businessId, productId]
  );
}

const crypto = require('node:crypto');
const fs = require('node:fs');
const path = require('node:path');

const DEFAULT_MIGRATION_DIRECTORY = path.resolve(
  __dirname,
  '../../database/migrations-mariadb'
);

const MIGRATION_LOCK_NAME = 'workflowos_schema_migrations';

function splitSqlStatements(sql) {
  const statements = [];

  let current = '';
  let quote = null;
  let lineComment = false;
  let blockComment = false;

  for (let i = 0; i < sql.length; i += 1) {
    const char = sql[i];
    const next = sql[i + 1];

    if (lineComment) {
      current += char;

      if (char === '\n') lineComment = false;
      continue;
    }

    if (blockComment) {
      current += char;

      if (char === '*' && next === '/') {
        current += next;
        i += 1;
        blockComment = false;
      }

      continue;
    }

    if (quote) {
      current += char;

      if (char === '\\' && next) {
        current += next;
        i += 1;
        continue;
      }

      if (char === quote) {
        if (next === quote) {
          current += next;
          i += 1;
        } else {
          quote = null;
        }
      }

      continue;
    }

    if (char === "'" || char === '"' || char === '`') {
      quote = char;
      current += char;
      continue;
    }

    if (char === '-' && next === '-' && /\s/.test(sql[i + 2] || ' ')) {
      lineComment = true;
      current += char + next;
      i += 1;
      continue;
    }

    if (char === '#') {
      lineComment = true;
      current += char;
      continue;
    }

    if (char === '/' && next === '*') {
      blockComment = true;
      current += char + next;
      i += 1;
      continue;
    }

    if (char === ';') {
      if (hasExecutableSql(current)) {
        statements.push(current.trim());
      }

      current = '';
      continue;
    }

    current += char;
  }

  if (quote || blockComment) {
    throw new Error('Invalid SQL migration: unclosed quote or comment.');
  }

  if (hasExecutableSql(current)) {
    statements.push(current.trim());
  }

  return statements;
}

function hasExecutableSql(statement) {
  const withoutComments = statement
    .replace(/\/\*[\s\S]*?\*\//g, '')
    .replace(/(^|\s)--[^\r\n]*/g, '$1')
    .replace(/(^|\s)#[^\r\n]*/g, '$1');

  return withoutComments.trim().length > 0;
}

async function applyMigrations(
  db,
  migrationDirectory = DEFAULT_MIGRATION_DIRECTORY
) {
  if (!db || typeof db.getConnection !== 'function') {
    throw new Error('A MariaDB connection pool is required.');
  }

  if (!fs.existsSync(migrationDirectory)) {
    throw new Error(
      `MariaDB migration directory not found: ${migrationDirectory}`
    );
  }

  const files = fs.readdirSync(migrationDirectory)
    .filter((file) => file.endsWith('.sql'))
    .sort();

  const applied = [];
  const connection = await db.getConnection();

  let lockAcquired = false;

  try {
    const [lockRows] = await connection.query(
      'SELECT GET_LOCK(?, 10) AS acquired',
      [MIGRATION_LOCK_NAME]
    );

    lockAcquired = Number(lockRows[0]?.acquired) === 1;

    if (!lockAcquired) {
      throw new Error('Could not acquire the database migration lock.');
    }

    await connection.query(`
      CREATE TABLE IF NOT EXISTS schema_migrations (
        version VARCHAR(255) NOT NULL PRIMARY KEY,
        checksum CHAR(64) NOT NULL,
        applied_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
      ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
    `);

    for (const file of files) {
      const filePath = path.join(migrationDirectory, file);
      const sql = fs.readFileSync(filePath, 'utf8');
      const checksum = crypto
        .createHash('sha256')
        .update(sql)
        .digest('hex');

      const [existingRows] = await connection.execute(
        'SELECT checksum FROM schema_migrations WHERE version = ?',
        [file]
      );

      if (existingRows.length > 0) {
        if (existingRows[0].checksum !== checksum) {
          throw new Error(
            `Applied migration checksum changed: ${file}`
          );
        }

        continue;
      }

      const statements = splitSqlStatements(sql);

      for (const statement of statements) {
        await connection.query(statement);
      }

      await connection.execute(
        `INSERT INTO schema_migrations (version, checksum)
         VALUES (?, ?)`,
        [file, checksum]
      );

      applied.push(file);
      console.log(`Applied MariaDB migration: ${file}`);
    }

    return applied;
  } finally {
    if (lockAcquired) {
      try {
        await connection.query(
          'SELECT RELEASE_LOCK(?)',
          [MIGRATION_LOCK_NAME]
        );
      } catch (error) {
        console.error('Could not release migration lock:', error.message);
      }
    }

    connection.release();
  }
}

function escapeHtml(value) {
  return String(value ?? '').replace(/[&<>"']/g, (character) => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;',
  }[character]));
}

function money(cents, currency = 'ZAR') {
  try {
    return new Intl.NumberFormat('en-ZA', { style: 'currency', currency, maximumFractionDigits: 2 }).format((Number(cents) || 0) / 100);
  } catch {
    return `${currency} ${((Number(cents) || 0) / 100).toFixed(2)}`;
  }
}

function dateTime(value, timezone) {
  if (!value) return '—';
  return new Intl.DateTimeFormat('en-ZA', { timeZone: timezone, dateStyle: 'medium', timeStyle: 'short' }).format(new Date(value));
}

async function api(url, options = {}) {
  const response = await fetch(url, {
    ...options,
    headers: {
      ...(options.body ? { 'Content-Type': 'application/json' } : {}),
      ...(sessionToken ? { Authorization: `Bearer ${sessionToken}` } : {}),
      ...options.headers,
    },
  });
  if (response.status === 401) {
    clearSession();
    throw new Error('Your session expired. Please sign in again.');
  }
  const body = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(body.error || 'The request could not be completed.');
  return body;
}

async function login(event) {
  event.preventDefault();
  const message = document.getElementById('auth-message');
  const submit = event.currentTarget.querySelector('button[type="submit"]');
  submit.disabled = true;
  submit.textContent = 'Signing in…';
  try {
    const data = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(Object.fromEntries(new FormData(event.currentTarget).entries())),
    }).then(async (response) => {
      const body = await response.json();
      if (!response.ok) throw new Error(body.error || 'Sign in failed.');
      return body;
    });
    sessionToken = data.sessionToken;
    localStorage.setItem('workflowos_session', sessionToken);
    history.replaceState({}, '', '/dashboard');
    await loadAuthenticatedApp();
  } catch (error) {
    message.textContent = error.message;
    message.className = 'message error';
    submit.disabled = false;
    submit.innerHTML = 'Sign in <span aria-hidden="true">→</span>';
  }
}

  }
}

async function loadAuthenticatedApp() {
  if (!sessionToken) {
    history.replaceState({}, '', '/login');
    renderAuth();
    return;
  }
  try {
    sessionInfo = await api('/api/session');
    if (!sessionInfo.business) throw new Error('No business is assigned to this account.');
    if (location.pathname !== '/dashboard' && location.pathname !== '/') history.replaceState({}, '', '/dashboard');
    if (location.pathname === '/') history.replaceState({}, '', '/dashboard');
    currentView = ['dashboard', 'sales', 'products', 'inventory', 'customers', 'settings'].includes(currentView) ? currentView : 'dashboard';
    renderShell();
    await renderView(currentView);
  } catch (error) {
    if (sessionToken) {
      clearSession();
      history.replaceState({}, '', '/login');
      renderAuth(error.message);
    }
  }
}

async function renderSales(content) {
  const [customers, data] = await Promise.all([api('/api/customers'), api('/api/dashboard')]);
  const currency = sessionInfo.business.currency;
  const cart = new Map();
  let selectedProduct = null;
  let searchTimer = null;
  let lastScannedBarcode = '';
  let lastScanAt = 0;

  content.innerHTML = `<section class="page-heading"><div><span class="eyebrow">REVENUE</span><h1>Sales & transactions</h1><p>Search your catalogue or scan product barcodes into a multi-item sale.</p></div></section>
    <div class="sales-layout"><section class="panel sales-entry"><div class="panel-heading"><div><h2>Build a transaction</h2><p>USB scanners act as keyboard input. Scan, or type a barcode and press Enter.</p></div></div>
      <label class="scanner-label">Scan barcode or search products<input id="product-lookup" autocomplete="off" placeholder="Scan barcode, search name or SKU…" aria-describedby="scan-feedback" /><small>Focus stays here after each successful scan.</small></label><div id="scan-feedback" class="scan-feedback" role="status" aria-live="polite"></div><div id="product-search-results" class="product-search-results"></div>
      <div id="selected-product" class="selected-product hidden"></div><button class="button button-primary add-selected-button hidden" id="add-selected-product" type="button">Add selected product</button>
      <div class="cart-heading"><h3>Current sale</h3><span id="cart-count">0 items</span></div><div id="sale-cart" class="sale-cart"><div class="cart-empty">Scan or search for a product to begin.</div></div>
      <form id="checkout-form" class="checkout-form"><label>Customer<select name="customerId"><option value="">Walk-in customer</option>${customers.customers.map((customer) => `<option value="${customer.id}">${escapeHtml(customer.name)}</option>`).join('')}</select></label><label>Payment method<select name="paymentMethod"><option value="cash">Cash</option><option value="card">Card</option><option value="eft">EFT</option><option value="other">Other</option></select></label><div class="cart-total"><span>Transaction total</span><strong id="cart-total">${money(0, currency)}</strong></div><button class="button button-primary button-wide" type="submit" id="complete-sale" disabled>Record completed sale</button></form>
      <p class="form-help">This records a transaction stated as already completed; it does not charge or verify a payment. Sale snapshots and stock movements are saved together. Only verified barcode mappings are used for automatic scans.</p>
    </section><section class="panel activity-panel"><div class="panel-heading"><div><h2>Today’s transactions</h2><p>${data.recentSales.length} transaction${data.recentSales.length === 1 ? '' : 's'} · ${escapeHtml(data.todayDate)} local time</p></div></div>${data.recentSales.length ? `<div class="table-scroll"><table><thead><tr><th>Reference</th><th>Time</th><th>Customer</th><th>Items</th><th>Amount</th><th>Payment</th><th>Status</th><th>Action</th></tr></thead><tbody>${data.recentSales.map((sale) => `<tr><td><strong>${escapeHtml(sale.reference)}</strong></td><td>${escapeHtml(dateTime(sale.created_at, data.business.timezone))}</td><td>${escapeHtml(sale.customer_name || 'Walk-in')}</td><td>${sale.items_count}</td><td><strong>${money(sale.total_cents, currency)}</strong></td><td>${escapeHtml(sale.payment_method || '—')}</td><td><span class="status-pill status-${escapeHtml(sale.status)}">${escapeHtml(sale.status)}</span></td><td>${sale.status === 'completed' ? `<button class="text-link refund-button" data-sale-id="${sale.id}">Refund & restock</button>` : '—'}</td></tr>`).join('')}</tbody></table></div>` : '<div class="table-empty"><strong>No transactions today</strong><span>Completed sales will be listed here.</span></div>'}</section></div>`;

  const input = document.getElementById('product-lookup');
  const feedback = document.getElementById('scan-feedback');
  const results = document.getElementById('product-search-results');
  const selectedPanel = document.getElementById('selected-product');
  const addSelected = document.getElementById('add-selected-product');

  function updateCart() {
    const rows = [...cart.values()];
    document.getElementById('cart-count').textContent = `${rows.reduce((sum, item) => sum + item.quantity, 0)} item${rows.length === 1 && rows[0]?.quantity === 1 ? '' : 's'}`;
    document.getElementById('cart-total').textContent = money(rows.reduce((sum, item) => sum + item.price_cents * item.quantity, 0), currency);
    document.getElementById('complete-sale').disabled = rows.length === 0;
    document.getElementById('sale-cart').innerHTML = rows.length ? rows.map((item) => `<div class="cart-row"><span class="product-initial">${escapeHtml(item.name.slice(0, 1).toUpperCase())}</span><span class="cart-product"><strong>${escapeHtml(item.name)}</strong><small>${escapeHtml(item.barcode || item.sku || 'No barcode/SKU')} · ${money(item.price_cents, currency)} each · ${item.quantity_on_hand} in stock</small></span><label class="cart-quantity-label">Qty<input class="cart-quantity" type="number" min="1" max="${item.quantity_on_hand}" value="${item.quantity}" data-product-id="${item.id}" aria-label="Quantity of ${escapeHtml(item.name)}" /></label><strong class="cart-line-total">${money(item.price_cents * item.quantity, currency)}</strong><button class="remove-cart-item" type="button" data-product-id="${item.id}" aria-label="Remove ${escapeHtml(item.name)}">×</button></div>`).join('') : '<div class="cart-empty">Scan or search for a product to begin.</div>';
    document.querySelectorAll('.cart-quantity').forEach((field) => field.addEventListener('change', () => {
      const item = cart.get(Number(field.dataset.productId));
      const quantity = Number(field.value);
      if (!item || !Number.isSafeInteger(quantity) || quantity < 1 || quantity > item.quantity_on_hand) { field.value = item?.quantity || 1; return; }
      item.quantity = quantity;
      updateCart();
    }));
    document.querySelectorAll('.remove-cart-item').forEach((button) => button.addEventListener('click', () => { cart.delete(Number(button.dataset.productId)); updateCart(); input.focus(); }));
  }

  function showSelected(product) {
    selectedProduct = product;
    selectedPanel.classList.remove('hidden');
    addSelected.classList.remove('hidden');
    selectedPanel.innerHTML = `<div class="selected-product-heading"><span class="product-initial">${escapeHtml(product.name.slice(0, 1).toUpperCase())}</span><div><strong>${escapeHtml(product.name)}</strong><small>${escapeHtml(product.brand || 'Business product')}</small></div></div><div class="selected-product-details"><span><small>Barcode ID</small><strong>${escapeHtml(product.barcode || 'Not recorded')}</strong></span><span><small>SKU</small><strong>${escapeHtml(product.sku || '—')}</strong></span><span><small>Category</small><strong>${escapeHtml(product.category || '—')}</strong></span><span><small>Selling price</small><strong>${money(product.price_cents, currency)}</strong></span><span><small>Available stock</small><strong>${product.quantity_on_hand}</strong></span><span><small>Barcode status</small><strong>${escapeHtml(product.barcode_status || 'unverified')}</strong></span></div>`;
    const inCart = cart.get(product.id)?.quantity || 0;
    addSelected.disabled = product.quantity_on_hand <= inCart;
    addSelected.textContent = product.quantity_on_hand <= inCart ? 'No more stock available' : 'Add 1 to sale';
  }

  function addToCart(product, scanner = false) {
    const existing = cart.get(product.id);
    if (!product.is_active || product.quantity_on_hand <= (existing?.quantity || 0)) {
      feedback.textContent = product.is_active ? `No available stock for ${product.name}.` : `${product.name} is inactive.`;
      feedback.className = 'scan-feedback error';
      return;
    }
    if (existing) existing.quantity += 1;
    else cart.set(product.id, { ...product, quantity: 1 });
    feedback.textContent = scanner ? `Added ${product.name} · ${money(product.price_cents, currency)} · ${product.quantity_on_hand - (existing ? existing.quantity : 1)} remaining after sale.` : `${product.name} added to this sale.`;
    feedback.className = 'scan-feedback success';
    results.innerHTML = '';
    selectedPanel.classList.add('hidden');
    addSelected.classList.add('hidden');
    selectedProduct = null;
    input.value = '';
    updateCart();
    input.focus();
  }

  async function lookupBarcode(value) {
    const now = Date.now();
    if (value === lastScannedBarcode && now - lastScanAt < 700) {
      feedback.textContent = 'Duplicate scan ignored. Scan again to add another unit.';
      feedback.className = 'scan-feedback neutral';
      return;
    }
    lastScannedBarcode = value;
    lastScanAt = now;
    try {
      const data = await api(`/api/products/barcode/${encodeURIComponent(value)}`);
      showSelected(data.product);
      addToCart(data.product, true);
    } catch (error) {
      feedback.textContent = error.message.includes('Invalid') ? 'Invalid barcode format or check digit.' : error.message.includes('unavailable') ? error.message : 'Barcode not found as a verified product for this business. Search by name or SKU, or add an unverified product to the business catalogue.';
      feedback.className = 'scan-feedback error';
      input.select();
      input.focus();
    }
  }

  async function searchProducts(query) {
    if (!query.trim()) { results.innerHTML = ''; selectedPanel.classList.add('hidden'); addSelected.classList.add('hidden'); selectedProduct = null; return; }
    try {
      const data = await api(`/api/products/search?q=${encodeURIComponent(query)}`);
      results.innerHTML = data.products.length ? data.products.map((product) => `<button class="search-result" type="button" data-product-id="${product.id}"><span class="product-initial">${escapeHtml(product.name.slice(0, 1).toUpperCase())}</span><span><strong>${escapeHtml(product.name)}</strong><small>${escapeHtml(product.sku || 'No SKU')} · ${escapeHtml(product.barcode || 'No barcode')} · stock ${product.quantity_on_hand}</small></span><strong>${money(product.price_cents, currency)}</strong></button>`).join('') : '<div class="search-no-results">No matching active products found.</div>';
      results.querySelectorAll('.search-result').forEach((button) => button.addEventListener('click', () => {
        const product = data.products.find((item) => item.id === Number(button.dataset.productId));
        if (product) showSelected(product);
      }));
    } catch (error) {
      feedback.textContent = error.message;
      feedback.className = 'scan-feedback error';
    }
  }

  input.addEventListener('input', () => {
    clearTimeout(searchTimer);
    searchTimer = setTimeout(() => searchProducts(input.value), 160);
  });
  input.addEventListener('keydown', async (event) => {
    if (event.key !== 'Enter') return;
    event.preventDefault();
    const value = input.value.trim();
    if (!value) return;
    if (/^\d+$/.test(value)) {
      clearTimeout(searchTimer);
      await lookupBarcode(value);
      return;
    }
    const firstResult = results.querySelector('.search-result');
    if (firstResult) firstResult.click();
    else await searchProducts(value);
  });
  addSelected.addEventListener('click', () => { if (selectedProduct) addToCart(selectedProduct); });

  document.getElementById('checkout-form').addEventListener('submit', async (event) => {
    event.preventDefault();
    const items = [...cart.values()].map((item) => ({ productId: item.id, quantity: item.quantity }));
    if (!items.length) return;
    const form = event.currentTarget;
    const values = Object.fromEntries(new FormData(form).entries());
    const submit = document.getElementById('complete-sale');
    submit.disabled = true;
    submit.textContent = 'Saving transaction…';
    try {
      await api('/api/sales', {
        method: 'POST',
        headers: { 'Idempotency-Key': crypto.randomUUID() },
        body: JSON.stringify({ paymentMethod: values.paymentMethod, customerId: values.customerId || null, items }),
      });
      currentView = 'sales';
      await renderView('sales');
      document.getElementById('scan-feedback').textContent = 'Transaction recorded; stock and dashboard source records updated.';
      document.getElementById('scan-feedback').className = 'scan-feedback success';
      document.getElementById('product-lookup').focus();
    } catch (error) {
      showInlineError(form, error.message);
      submit.disabled = false;
      submit.textContent = 'Record completed sale';
    }
  });
  content.querySelectorAll('.refund-button').forEach((button) => button.addEventListener('click', async () => {
    if (!confirm('Refund this completed sale and return its quantities to stock?')) return;
    button.disabled = true;
    try { await api(`/api/sales/${button.dataset.saleId}/refund`, { method: 'POST' }); await renderView('sales'); }
    catch (error) { alert(error.message); button.disabled = false; }
  }));
  input.focus();
}

function showInlineError(form, message) {
  let feedback = form.querySelector('.inline-feedback');
  if (!feedback) {
    feedback = document.createElement('p');
    feedback.className = 'inline-feedback error';
    form.append(feedback);
  }
  feedback.textContent = message;
}

function identifyBarcodeFormat(value) {
  const barcode = String(value ?? '').trim();
  if (!/^\d+$/.test(barcode)) return null;
  if (barcode.length === 13) return 'EAN-13';
  if (barcode.length === 8) return 'EAN-8';
  if (barcode.length === 12) return 'UPC-A';
  return null;
}

function isValidBarcode(value) {
  const barcode = String(value ?? '').trim();
  const format = identifyBarcodeFormat(barcode);
  if (!format) return false;

  const digits = [...barcode].map(Number);
  const checkDigit = digits.pop();
  const sum = digits.reduce((total, digit, index) => {
    const weight = (digits.length - 1 - index) % 2 === 0 ? 3 : 1;
    return total + digit * weight;
  }, 0);

  return (10 - (sum % 10)) % 10 === checkDigit;
}

function normalizeBarcode(value) {
  const barcode = String(value ?? '').trim();
  if (!/^\d+$/.test(barcode)) throw new Error('INVALID_BARCODE');
  const format = identifyBarcodeFormat(barcode);
  if (!format || !isValidBarcode(barcode)) throw new Error('INVALID_BARCODE');
  return { barcode, format };
}

module.exports = {
  identifyBarcodeFormat,
  isValidBarcode,
  normalizeBarcode,
};

async function queryOne(sql, params = [], connection = db) {
  const [rows] = await connection.execute(sql, params);
  return rows[0] || null;

async function queryMany(sql, params = [], connection = db) {
  const [rows] = await connection.execute(sql, params);
  return rows;
}

async function execute(sql, params = [], connection = db) {
  const [result] = await connection.execute(sql, params);
  return result;
}

async function withTransaction(callback) {
  const connection = await db.getConnection();

  try {
    await connection.beginTransaction();
    const result = await callback(connection);
    await connection.commit();
    return result;
  } catch (error) {
    await connection.rollback();
    throw error;
  } finally {
    connection.release();
  }
}

function parseCsvLine(line) {
  const fields = [];
  let value = '';
  let quoted = false;

  for (let index = 0; index < line.length; index += 1) {
    const character = line[index];

    if (
      character === '"' &&
      quoted &&
      line[index + 1] === '"'
    ) {
      value += '"';
      index += 1;
    } else if (character === '"') {
      quoted = !quoted;
    } else if (character === ',' && !quoted) {
      fields.push(value);
      value = '';
    } else {
      value += character;
    }
  }

  if (quoted) throw new Error('INVALID_CSV');

  fields.push(value);
  return fields;
}

async function importProductsCsv({
  businessId,
  userId,
  csv,
}) {
  await requireMembership(businessId, userId);

  if (
    typeof csv !== 'string' ||
    csv.length > 1024 * 1024
  ) {
    throw new Error('INVALID_CSV');
  }

  const lines = csv
    .replace(/^\uFEFF/, '')
    .split(/\r?\n/)
    .filter((line) => line.trim());

  if (lines.length < 2) throw new Error('INVALID_CSV');

  const headers = parseCsvLine(lines.shift())
    .map((header) =>
      header.trim().toLowerCase().replace(/[_\s-]/g, '')
    );

  const column = (name) =>
    headers.indexOf(
      name.toLowerCase().replace(/[_\s-]/g, '')
    );

  if (column('name') < 0) throw new Error('INVALID_CSV');

  const summary = {
    imported: 0,
    duplicates: 0,
    invalid: 0,
    incomplete: 0,
    errors: [],
  };

  await withTransaction(async (connection) => {
    for (const [index, line] of lines.entries()) {
      let cells;

      try {
        cells = parseCsvLine(line);
      } catch {
        summary.invalid += 1;
        summary.errors.push({
          row: index + 2,
          reason: 'Malformed CSV row',
        });
        continue;
      }

      const get = (name) =>
        column(name) >= 0
          ? (cells[column(name)] || '').trim()
          : '';

      const name = get('name');
      const barcodeValue = get('barcode');
      const sku = get('sku') || null;
      const price = Number(get('price') || 0);
      const costText = get('cost');
      const cost = Number(costText || 0);
      const quantity = Number(get('initialstock') || 0);
      const reorder = Number(get('reorderlevel') || 0);
      const unitOfMeasure = get('unitofmeasure') || 'unit';
      const imageUrl = get('imageurl') || null;

      if (imageUrl && !isValidHttpsUrl(imageUrl)) {
        summary.invalid += 1;
        summary.errors.push({
          row: index + 2,
          reason: 'Image URL must use HTTPS',
        });
        continue;
      }

      if (
        !name ||
        !Number.isFinite(price) ||
        price < 0 ||
        !Number.isFinite(cost) ||
        cost < 0 ||
        !Number.isSafeInteger(Math.round(price * 100)) ||
        !Number.isSafeInteger(Math.round(cost * 100)) ||
        !Number.isSafeInteger(quantity) ||
        quantity < 0 ||
        !Number.isSafeInteger(reorder) ||
        reorder < 0 ||
        !unitOfMeasure ||
        unitOfMeasure.length > 32
      ) {
        summary.invalid += 1;
        summary.errors.push({
          row: index + 2,
          reason: 'Invalid name, price, cost, or stock value',
        });
        continue;
      }

      if (sku) {
        const duplicateSku = await queryOne(
          `
          SELECT id
          FROM products
          WHERE business_id = ? AND sku = ?
          `,
          [businessId, sku],
          connection
        );

        if (duplicateSku) {
          summary.duplicates += 1;
          continue;
        }
      }

      let barcode = null;

      if (barcodeValue) {
        try {
          barcode = normalizeBarcode(barcodeValue).barcode;
        } catch {
          summary.invalid += 1;
          summary.errors.push({
            row: index + 2,
            reason: 'Barcode is not a valid EAN-13, EAN-8, or UPC-A check-digit value',
          });
          continue;
        }

        const duplicateBarcode = await queryOne(
          `
          SELECT id
          FROM products
          WHERE business_id = ? AND barcode = ?
          `,
          [businessId, barcode],
          connection
        );

        if (duplicateBarcode) {
          summary.duplicates += 1;
          continue;
        }
      }

      const result = await execute(
        `
        INSERT INTO products (
          business_id,
          name,
          brand,
          barcode,
          barcode_status,
          sku,
          category,
          description,
          packaging_size,
          unit_of_measure,
          image_url,
          price_cents,
          cost_cents,
          cost_is_known,
          is_active
        )
        VALUES (?, ?, ?, ?, 'unverified', ?, ?, ?, ?, ?, ?, ?, ?, ?, 1)
        `,
        [
          businessId,
          name,
          get('brand') || null,
          barcode,
          sku,
          get('category') || null,
          get('description') || null,
          get('packagingsize') || null,
          unitOfMeasure,
          imageUrl,
          Math.round(price * 100),
          Math.round(cost * 100),
          costText ? 1 : 0,
        ],
        connection
      );

      const productId = Number(result.insertId);

      await execute(
        `
        INSERT INTO inventory (
          business_id,
          product_id,
          quantity_on_hand,
          reorder_level
        )
        VALUES (?, ?, ?, ?)
        `,
        [businessId, productId, quantity, reorder],
        connection
      );

      if (quantity > 0) {
        await execute(
          `
          INSERT INTO inventory_movements (
            business_id,
            product_id,
            movement_type,
            quantity,
            reference_type,
            reference_id,
            notes,
            created_at
          )
          VALUES (?, ?, 'opening', ?, 'product', ?, 'Opening stock from CSV import', ?)
          `,
          [
            businessId,
            productId,
            quantity,
            productId,
            currentDateTime(),
          ],
          connection
        );
      }

      summary.imported += 1;
      summary.incomplete += 1;
    }
  });

  return summary;
}

function numeric(value) {
  const result = Number(value ?? 0);
  return Number.isFinite(result) ? result : 0;
}

// Convert an ISO date string into a JavaScript Date.
// mysql2 serializes Date parameters as SQL-compatible
// date/time values when used in prepared statements.
function dbDate(value) {
  return value instanceof Date ? value : new Date(value);
}

function datePartsInZone(date, timeZone) {
  const parts = new Intl.DateTimeFormat('en-GB', {
    timeZone,
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
    hour: '2-digit',
    minute: '2-digit',
    second: '2-digit',
    hourCycle: 'h23',
  }).formatToParts(date);

  return Object.fromEntries(
    parts
      .filter((part) => part.type !== 'literal')
      .map((part) => [part.type, Number(part.value)])
  );
}

function localDateAt(date, timeZone) {
  const parts = datePartsInZone(date, timeZone);

  return `${parts.year}-${String(parts.month).padStart(2, '0')}-${String(parts.day).padStart(2, '0')}`;
}

function addDays(dateString, days) {
  const date = new Date(`${dateString}T00:00:00.000Z`);
  date.setUTCDate(date.getUTCDate() + days);
  return date.toISOString().slice(0, 10);
}

function zonedMidnightUtc(dateString, timeZone) {
  const targetMs = Date.parse(`${dateString}T00:00:00.000Z`);
  let guessMs = targetMs;

  for (let attempt = 0; attempt < 4; attempt += 1) {
    const local = datePartsInZone(new Date(guessMs), timeZone);

    const representedMs = Date.UTC(
      local.year,
      local.month - 1,
      local.day,
      local.hour,
      local.minute,
      local.second
    );

    const adjustment = targetMs - representedMs;
    guessMs += adjustment;

    if (adjustment === 0) break;
  }

  return new Date(guessMs).toISOString();
}

function dayBounds(dateString, timeZone) {
  return {
    start: zonedMidnightUtc(dateString, timeZone),
    end: zonedMidnightUtc(addDays(dateString, 1), timeZone),
  };
}

function percentageChange(current, previous) {
  if (!previous) return null;

  return Math.round(
    ((current - previous) / previous) * 10000
  ) / 100;
}

async function aggregateSales(businessId, start, end) {
  const [sales, items] = await Promise.all([
    queryOne(
      `
      SELECT
        COUNT(*) AS transaction_count,
        COALESCE(SUM(total_cents), 0) AS revenue_cents
      FROM sales
      WHERE business_id = ?
        AND status = 'completed'
        AND created_at >= ?
        AND created_at < ?
      `,
      [businessId, dbDate(start), dbDate(end)]
    ),

    queryOne(
      `
      SELECT
        COALESCE(SUM(si.quantity), 0) AS items_sold,
        COUNT(DISTINCT si.product_id) AS unique_products,
        COALESCE(
          SUM(
            CASE
              WHEN si.unit_cost_cents IS NOT NULL
              THEN si.line_total_cents - (si.quantity * si.unit_cost_cents)
              ELSE 0
            END
          ),
          0
        ) AS gross_profit_cents,
        COALESCE(
          SUM(
            CASE
              WHEN si.unit_cost_cents IS NULL
              THEN si.quantity
              ELSE 0
            END
          ),
          0
        ) AS unknown_cost_quantity
      FROM sale_items si
      JOIN sales s ON s.id = si.sale_id
      WHERE s.business_id = ?
        AND s.status = 'completed'
        AND s.created_at >= ?
        AND s.created_at < ?
      `,
      [businessId, dbDate(start), dbDate(end)]
    ),
  ]);

  const revenueCents = numeric(sales?.revenue_cents);
  const transactionCount = numeric(sales?.transaction_count);
  const itemsSold = numeric(items?.items_sold);
  const grossProfitCents = numeric(items?.gross_profit_cents);
  const unknownCostQuantity = numeric(items?.unknown_cost_quantity);

  return {
    revenueCents,
    transactionCount,
    averageTransactionCents: transactionCount
      ? Math.round(revenueCents / transactionCount)
      : 0,
    itemsSold,
    uniqueProductsSold: numeric(items?.unique_products),
    grossProfitCents,
    grossProfitComplete: !unknownCostQuantity,
    grossProfitMarginPct:
      revenueCents && !unknownCostQuantity
        ? Math.round((grossProfitCents / revenueCents) * 10000) / 100
        : null,
  };
}

function getPeriodBounds(
  period,
  timeZone,
  now,
  customStart,
  customEnd
) {
  const today = localDateAt(now, timeZone);

  if (period === 'custom') {
    if (
      !/^\d{4}-\d{2}-\d{2}$/.test(customStart || '') ||
      !/^\d{4}-\d{2}-\d{2}$/.test(customEnd || '') ||
      customStart > customEnd
    ) {
      throw new Error('INVALID_DATE_RANGE');
    }

    const isRealDate = (date) => {
      const parsed = new Date(`${date}T00:00:00.000Z`);

      return (
        !Number.isNaN(parsed.getTime()) &&
        parsed.toISOString().slice(0, 10) === date
      );
    };

    if (!isRealDate(customStart) || !isRealDate(customEnd)) {
      throw new Error('INVALID_DATE_RANGE');
    }

    const days = (
      Date.parse(`${customEnd}T00:00:00Z`) -
      Date.parse(`${customStart}T00:00:00Z`)
    ) / 86400000;

    if (days > 365) throw new Error('INVALID_DATE_RANGE');

    return {
      startDate: customStart,
      endDate: addDays(customEnd, 1),
      period,
    };
  }

  if (period === 'today') {
    return {
      startDate: today,
      endDate: addDays(today, 1),
      period,
    };
  }

  if (period === '30d') {
    return {
      startDate: addDays(today, -29),
      endDate: addDays(today, 1),
      period,
    };
  }

  if (period === 'month') {
    return {
      startDate: `${today.slice(0, 7)}-01`,
      endDate: addDays(today, 1),
      period,
    };
  }

  return {
    startDate: addDays(today, -6),
    endDate: addDays(today, 1),
    period: '7d',
  };
}

function getRevenueTrends(
  sales,
  startDate,
  endDate,
  timeZone,
  granularity
) {
  const selectedGranularity = [
    'daily',
    'weekly',
    'monthly',
  ].includes(granularity)
    ? granularity
    : 'daily';

  const buckets = new Map();

  for (
    let day = startDate;
    day < endDate;
    day = addDays(day, 1)
  ) {
    let key = day;

    if (selectedGranularity === 'monthly') {
      key = day.slice(0, 7);
    }

    if (selectedGranularity === 'weekly') {
      const weekday = new Date(
        `${day}T00:00:00.000Z`
      ).getUTCDay();

      key = addDays(day, -((weekday + 6) % 7));
    }

    if (!buckets.has(key)) buckets.set(key, 0);
  }

  for (const sale of sales) {
    const day = localDateAt(
      new Date(sale.created_at),
      timeZone
    );

    let key = day;

    if (selectedGranularity === 'monthly') {
      key = day.slice(0, 7);
    }

    if (selectedGranularity === 'weekly') {
      const weekday = new Date(
        `${day}T00:00:00.000Z`
      ).getUTCDay();

      key = addDays(day, -((weekday + 6) % 7));
    }

    buckets.set(
      key,
      (buckets.get(key) || 0) + numeric(sale.total_cents)
    );
  }

  return [...buckets.entries()].map(
    ([date, revenueCents]) => ({
      date,
      revenueCents,
    })
  );
}

        revenueChangePct: percentageChange(
          today.revenueCents,
          previousDay.revenueCents
        ),
        transactionChangePct: percentageChange(
          today.transactionCount,
          previousDay.transactionCount
        ),
        itemsSoldChangePct: percentageChange(
          today.itemsSold,
          previousDay.itemsSold
        ),
        previousRevenueCents: previousDay.revenueCents,
        previousTransactionCount: previousDay.transactionCount,
        previousItemsSold: previousDay.itemsSold,
      },

      customers: {
        total: numeric(customerStats?.total),
        newToday: numeric(customerStats?.new_today),
        returningToday: numeric(returningCustomerStats?.total),
      },

      inventory: {
        lowStockCount: numeric(inventory?.low_stock_count),
        outOfStockCount: numeric(inventory?.out_of_stock_count),
        lowStockProducts,
      },

      productCount: totalProducts,
    },

    bestSellers,

    recentSales,

    trends,

    insights: {
      mostProfitableProduct: bestProfit
        ? {
            name: bestProfit.name,
            grossProfitCents: bestProfit.grossProfitCents,
          }
        : null,

      bestCategory: category
        ? {
            name: category.name,
            revenueCents: numeric(category.revenue_cents),
          }
        : null,

      productsNotSoldInPeriod: Math.max(
        0,
        totalProducts - soldProducts.size
      ),

      averageTransactionCents: today.averageTransactionCents,

      costCoverageComplete: today.grossProfitComplete,
    },
  };
}
